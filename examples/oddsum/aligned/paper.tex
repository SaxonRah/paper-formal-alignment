Throughout, $\mathbb{N} = \{0, 1, 2, \dots\}$.

\begin{definition}\formal{OddSum.S}
For $n \in \mathbb{N}$ let $S(n)$ be the sum of the first $n$ odd numbers:
$S(0) = 0$ and $S(n+1) = S(n) + (2n+1)$.
\end{definition}

\begin{example}\formal{OddSum.S_three}
\concl{S 3 = 9}{$S(3) = 1 + 3 + 5 = 9$}.
\end{example}

\begin{thm}\label{thm:sq}\formal{OddSum.S_eq_sq}
For every \var{n}{nat}{$n \in \mathbb{N}$}, \concl{S n = n ^ 2}{$S(n) = n^2$}.
\end{thm}

\begin{thm}\label{thm:even}\formal{OddSum.S_even}
If \hyp{h_even}{exists k : nat, n = 2 * k}{\var{n}{nat}{$n$} is even}, then \concl{exists k : nat, S n = 2 * k}{$S(n)$ is even}.
\end{thm}

\begin{thm}\label{thm:diff}\formal{OddSum.S_diff}
For every \hyp{h_pos}{1 <= n}{$\var{n}{nat}{n} \geq 1$}, \concl{S n - S (n - 1) = 2 * n - 1}{$S(n) - S(n-1) = 2n - 1$}.
\end{thm}

\begin{cor}
$S(n)$ is odd if and only if $n$ is odd.
\end{cor}
